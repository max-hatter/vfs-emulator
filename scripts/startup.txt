# Стартовый скрипт для тестирования эмулятора
# Выполняется последовательно, ошибочные строки пропускаются

# Приветствие
echo Hello from startup script

# Заглушки команд
ls
ls -la /tmp
cd /home
cd $HOME

# Демонстрация ошибок (строки будут пропущены, скрипт продолжится)
unknown_command
cd
exit_with_args foo

# Проверка раскрытия переменных окружения
echo $USERPROFILE
echo $PATH

# Завершение
exit
